# EP09-P4: Recovery Group Mobile UI & i18n

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox syntax for tracking.

**Goal:** Implement recovery group admin and user flows on iOS (SwiftUI) and Android (Kotlin/Compose) with full parity to desktop, plus i18n strings across all 13 locales.

**Architecture:** Mobile platforms wrap the same Rust `packages/crypto` crate via UniFFI (iOS XCFramework) and JNI (Android `.so`) for all Shamir operations. API calls use the existing `APIService` (iOS URLSession) and `ApiService` (Android OkHttp) patterns. Views follow established admin panel patterns: iOS uses `@Observable` ViewModels with `@Bindable` bindings and `@Environment(HubContext.self)` for hub scoping; Android uses Hilt-injected ViewModels with `StateFlow` and Compose `collectAsState()`. All user-facing strings go through `packages/i18n` with codegen to iOS `.strings` and Android `strings.xml`.

**Tech Stack:** SwiftUI (iOS 17+), Kotlin/Compose (Material 3, Hilt), packages/i18n codegen

```swift
// MARK: - Shamir Secret Sharing

/// Split a secret into Shamir shares.
/// - Parameters:
///   - secretHex: The secret to split (hex-encoded bytes).
///   - total: Total number of shares (3-5).
///   - threshold: Minimum shares needed to reconstruct (2-5, must be <= total).
/// - Returns: Array of ShamirShare objects.
func shamirSplit(secretHex: String, total: UInt8, threshold: UInt8) throws -> [ShamirShare] {
    try ffiMobileShamirSplit(secretHex: secretHex, total: total, threshold: threshold)
}

/// Combine Shamir shares to reconstruct the original secret.
/// - Parameter shares: Array of ShamirShare objects (must have >= threshold shares).
/// - Returns: The reconstructed secret (hex-encoded bytes).
func shamirCombine(shares: [ShamirShare]) throws -> String {
    let encoder = JSONEncoder()
    let data = try encoder.encode(shares)
    guard let json = String(data: data, encoding: .utf8) else {
        throw CryptoServiceError.encryptionFailed("Failed to encode shares to JSON")
    }
    return try ffiMobileShamirCombine(sharesJson: json)
}

/// Compute a SHA-256 commitment for a Shamir share (for tamper detection).
/// - Parameter share: The share to commit.
/// - Returns: SHA-256 commitment hex string.
func shamirCommit(share: ShamirShare) throws -> String {
    let encoder = JSONEncoder()
    let data = try encoder.encode(share)
    guard let json = String(data: data, encoding: .utf8) else {
        throw CryptoServiceError.encryptionFailed("Failed to encode share to JSON")
    }
    return try ffiMobileShamirCommit(shareJson: json)
}

/// Verify a Shamir share against its SHA-256 commitment.
/// - Parameters:
///   - share: The share to verify.
///   - commitment: The expected commitment hex string.
/// - Returns: true if the share matches the commitment.
func shamirVerify(share: ShamirShare, commitment: String) throws -> Bool {
    let encoder = JSONEncoder()
    let data = try encoder.encode(share)
    guard let json = String(data: data, encoding: .utf8) else {
        throw CryptoServiceError.encryptionFailed("Failed to encode share to JSON")
    }
    return try ffiMobileShamirVerify(shareJson: json, commitmentHex: commitment)
}

// MARK: - Recovery Group

/// Generate a new X25519 keypair for a recovery group.
/// The private key will be Shamir-split and never stored whole.
/// - Returns: RecoveryGroupKeypair with publicKeyHex and privateKeyHex.
func recoveryGroupGenerateKeypair() throws -> RecoveryGroupKeypair {
    try ffiMobileRecoveryGroupGenerateKeypair()
}
```

**Note:** `ShamirShare` and `RecoveryGroupKeypair` are UniFFI-generated types from `packages/crypto`. They will be available after `bun run ios:xcframework` rebuilds the XCFramework with the new Shamir module. If UniFFI generates them as structs, they are already `Codable` via the UniFFI Swift bindings. If not, add `Codable` conformance manually.

```swift
// MARK: - Recovery Group API

/// Enroll a recovery group for the active hub.
func enrollRecoveryGroup(_ body: [String: Any]) async throws -> OkResponse {
    let jsonData = try JSONSerialization.data(withJSONObject: body)
    return try await request(method: "POST", path: hp("/api/recovery-group/enroll"), rawBody: jsonData)
}

/// Get recovery group config for a hub.
func getRecoveryGroup(hubId: String) async throws -> RecoveryGroupInfo {
    try await request(method: "GET", path: hp("/api/recovery-group/\(hubId)"))
}

/// Initiate account recovery (unauthenticated).
func initiateRecovery(hubId: String, userIdentifier: String, newDevicePubkey: String) async throws -> RecoveryInitiateResponse {
    let body: [String: String] = [
        "hubId": hubId,
        "userIdentifier": userIdentifier,
        "newDevicePubkey": newDevicePubkey,
    ]
    return try await request(method: "POST", path: "/api/recovery-group/initiate", body: body)
}

/// Verify Signal verification code for recovery initiation (unauthenticated).
func verifyRecoveryCode(sessionId: String, verificationCode: String) async throws -> RecoveryVerifyResponse {
    let body: [String: String] = [
        "sessionId": sessionId,
        "verificationCode": verificationCode,
    ]
    return try await request(method: "POST", path: "/api/recovery-group/initiate/verify", body: body)
}

/// Get recovery session status.
func getRecoverySession(sessionId: String) async throws -> RecoverySessionStatus {
    try await request(method: "GET", path: hp("/api/recovery-group/session/\(sessionId)"))
}
```

```swift
/// Permission-gated: requires `recovery:manage`.
struct RecoveryTeamConfigView: View {
    @Bindable var viewModel: AdminViewModel
    @Environment(HubContext.self) private var hubContext

    var body: some View {
        ZStack {
            if viewModel.isLoadingRecoveryGroup {
                loadingState
            } else if viewModel.recoveryGroupInfo == nil {
                setupState
            } else {
                configuredState
            }
        }
        .task(id: hubContext.activeHubId) {
            await viewModel.loadRecoveryGroup()
        }
        .alert(
            NSLocalizedString("recoveryGroup.requests.cancelConfirm", comment: ""),
            isPresented: $viewModel.showRotateConfirmation
        ) {

                Task { await viewModel.rotateRecoveryGroup() }
            }
            Button(NSLocalizedString("cancel", comment: "Cancel"), role: .cancel) {}
        }
    }

    // MARK: - Setup State (No Team Configured)

    private var setupState: some View {
        ScrollView {
            VStack(spacing: 24) {
                // Header
                VStack(spacing: 8) {
                    Image(systemName: "person.3.fill")
                        .font(.largeTitle)
                        .foregroundStyle(Color.brandPrimary)
                    Text(NSLocalizedString("recoveryGroup.title", comment: "Recovery Team"))
                        .font(.brand(.title2))
                        .fontWeight(.bold)
                    Text(NSLocalizedString("recoveryGroup.description", comment: ""))
                        .font(.brand(.body))
                        .foregroundStyle(Color.brandMutedForeground)
                        .multilineTextAlignment(.center)
                }
                .padding(.top, 32)

                // Configuration form
                recoveryConfigForm

                // Setup button
                Button {
                    Task { await viewModel.enrollRecoveryGroup() }
                } label: {
                    if viewModel.isEnrollingRecoveryGroup {
                        HStack(spacing: 8) {
                            ProgressView()
                                .tint(.white)
                            Text(NSLocalizedString("recoveryGroup.settingUp", comment: "Setting up..."))
                        }
                    } else {
                        Text(NSLocalizedString("recoveryGroup.setup", comment: "Set up recovery team"))
                    }
                }
                .buttonStyle(.borderedProminent)
                .disabled(!viewModel.isRecoveryConfigValid || viewModel.isEnrollingRecoveryGroup)
                .accessibilityIdentifier("setup-recovery-team-button")

                if let error = viewModel.recoveryGroupError {
                    Text(error)
                        .font(.brand(.footnote))

                // Status section
                Section {

                        NSLocalizedString("recoveryGroup.requiredApprovals", comment: ""),

                        NSLocalizedString("recoveryGroup.totalContacts", comment: ""),

                        NSLocalizedString("recoveryGroup.delayConfig", comment: ""),

                        LabeledContent(
                            NSLocalizedString("recoveryGroup.lastRotated", comment: ""),
```

```swift
    private func verifyCode() async {
        isSubmitting = true
        errorMessage = nil
        do {
            let _ = try await apiService.verifyRecoveryCode(
                sessionId: sessionId,
                verificationCode: verificationCode
            )
            step = .waiting
        } catch {
            errorMessage = NSLocalizedString("recoveryGroup.error.signalVerificationFailed", comment: "")
        }
        isSubmitting = false
    }

    private func startPolling() {
        pollTimer = Timer.scheduledTimer(withTimeInterval: 5.0, repeats: true) { _ in
            Task { await pollSessionStatus() }
        }
    }

    private func pollSessionStatus() async {
        do {
            let status = try await apiService.getRecoverySession(sessionId: sessionId)
            await MainActor.run {
                contributionCount = status.contributionCount
                requiredApprovals = status.threshold
                delayRemainingMs = status.delayRemainingMs ?? 0
                sessionStatus = status.status
                if status.status == "completed" {
                    pollTimer?.invalidate()
                    Task { await completeRecovery(status: status) }
                } else if ["expired", "cancelled"].contains(status.status) {
                    pollTimer?.invalidate()
                    errorMessage = NSLocalizedString("recoveryGroup.error.sessionExpired", comment: "")
                }
            }
        } catch {
            // Silently retry on network errors during polling
        }
    }

    private func completeRecovery(status: RecoverySessionStatus) async {
        guard let contributions = status.contributions else { return }
        do {
            // Decrypt each contribution (HPKE-open with device key)
            var shares: [ShamirShare] = []
            for contribution in contributions {
                let envelope = HpkeEnvelope(
                    v: 1,
                    labelId: 0, // Will be set by the crypto layer based on label
                    enc: "", // Parsed from contribution.encryptedShare
                    ct: contribution.encryptedShare
                )
                let shareHex = try cryptoService.hpkeOpen(
                    envelope: envelope,
                    expectedLabel: "llamenos:recovery-group:share-contribute:v1",
                    aadHex: ""
                )
                // Parse share from hex -- the exact format depends on UniFFI ShamirShare type
                // This is a simplified representation; actual parsing via UniFFI deserializer
                let shareData = Data(hexString: shareHex)
                let share = try JSONDecoder().decode(ShamirShare.self, from: shareData ?? Data())
                shares.append(share)
            }

            // Combine shares to reconstruct recovery group private key
            let recoveryPrivKeyHex = try cryptoService.shamirCombine(shares: shares)

            // Use recovery private key to decrypt user's PUK seed envelope
            // (PUK seed was HPKE-wrapped under recovery group pubkey)
            // This step requires fetching the user recovery envelope from the server
            // and HPKE-opening it with the reconstructed private key.

            // Zeroize recovery private key after use (handled by Rust FFI)
            step = .complete
        } catch {
            errorMessage = NSLocalizedString("recoveryGroup.error.commitmentFailed", comment: "")
        }
    }

    private func completeRecoveryWithPin() async {
        guard newPin == confirmPin, newPin.count >= 6 else {
            errorMessage = NSLocalizedString("error_invalid_pin", comment: "")
            return
        }
        isSubmitting = true
        // In practice: re-encrypt device keys with new PIN, store, dismiss
        // The device is already provisioned from submitRecoveryRequest()
        isSubmitting = false
        dismiss()
    }
}

// MARK: - Data+Hex extension

private extension Data {
    init?(hexString: String) {
        let hex = hexString
        guard hex.count.isMultiple(of: 2) else { return nil }
        var data = Data(capacity: hex.count / 2)
        var index = hex.startIndex
        while index < hex.endIndex {
            let nextIndex = hex.index(index, offsetBy: 2)
            guard let byte = UInt8(hex[index..<nextIndex], radix: 16) else { return nil }
            data.append(byte)
            index = nextIndex
        }
        self = data
    }
}
```

```kotlin
/**
 * Split a secret into Shamir shares.
 * @param secretHex The secret to split (hex-encoded bytes).
 * @param total Total number of shares (3-5).
 * @param threshold Minimum shares needed to reconstruct (2-5, must be <= total).
 * @return List of ShamirShare objects.
 */
suspend fun shamirSplit(secretHex: String, total: Int, threshold: Int): List<ShamirShare> =
    withContext(computeDispatcher) {
        check(nativeLibLoaded) { "Native crypto library not loaded." }
        try {
            val ffiShares = org.llamenos.core.mobileShamirSplit(
                secretHex = secretHex,
                total = total.toUByte(),
                threshold = threshold.toUByte(),
            )
            ffiShares.map { share ->
                ShamirShare(
                    x = share.x.toInt(),
                    y = share.y,
                )
            }
        } catch (e: Exception) {
            throw CryptoException("Shamir split failed: ${e.message}", e)
        }
    }

/**
 * Combine Shamir shares to reconstruct the original secret.
 * @param shares List of ShamirShare objects (must have >= threshold shares).
 * @return The reconstructed secret (hex-encoded bytes).
 */
suspend fun shamirCombine(shares: List<ShamirShare>): String =
    withContext(computeDispatcher) {
        check(nativeLibLoaded) { "Native crypto library not loaded." }
        try {
            val sharesJson = json.encodeToString(
                kotlinx.serialization.builtins.ListSerializer(
                    kotlinx.serialization.builtins.serializer<ShamirShare>()
                ),
                shares,
            )
            org.llamenos.core.mobileShamirCombine(sharesJson = sharesJson)
        } catch (e: Exception) {
            throw CryptoException("Shamir combine failed: ${e.message}", e)
        }
    }

/**
 * Compute a SHA-256 commitment for a Shamir share (tamper detection).
 * @param share The share to commit.
 * @return SHA-256 commitment hex string.
 */
suspend fun shamirCommit(share: ShamirShare): String =
    withContext(computeDispatcher) {
        check(nativeLibLoaded) { "Native crypto library not loaded." }
        try {
            val shareJson = json.encodeToString(ShamirShare.serializer(), share)
            org.llamenos.core.mobileShamirCommit(shareJson = shareJson)
        } catch (e: Exception) {
            throw CryptoException("Shamir commit failed: ${e.message}", e)
        }
    }

/**
 * Verify a Shamir share against its SHA-256 commitment.
 * @param share The share to verify.
 * @param commitment The expected commitment hex string.
 * @return true if the share matches the commitment.
 */
suspend fun shamirVerify(share: ShamirShare, commitment: String): Boolean =
    withContext(computeDispatcher) {
        check(nativeLibLoaded) { "Native crypto library not loaded." }
        try {
            val shareJson = json.encodeToString(ShamirShare.serializer(), share)
            org.llamenos.core.mobileShamirVerify(
                shareJson = shareJson,
                commitmentHex = commitment,
            )
        } catch (e: Exception) {
            throw CryptoException("Shamir verify failed: ${e.message}", e)
        }
    }

 * The private key will be Shamir-split and never stored whole.
```

```kotlin
/**
 * Unauthenticated user flow for recovering an account after device loss.
 * Presented from the login screen via "I lost my device" link.
 */
@Composable
fun AccountRecoveryScreen(
    apiService: ApiService,
    cryptoService: CryptoService,
    onRecoveryComplete: () -> Unit,
    onCancel: () -> Unit,
    modifier: Modifier = Modifier,
) {
    var step by remember { mutableStateOf(RecoveryStep.IDENTIFIER) }
    var userIdentifier by remember { mutableStateOf("") }
    var hubId by remember { mutableStateOf("") }
    var sessionId by remember { mutableStateOf("") }
    var verificationCode by remember { mutableStateOf("") }
    var newPin by remember { mutableStateOf("") }
    var confirmPin by remember { mutableStateOf("") }
    var isSubmitting by remember { mutableStateOf(false) }
    var errorMessage by remember { mutableStateOf<String?>(null) }
    var contributionCount by remember { mutableIntStateOf(0) }
    var requiredApprovals by remember { mutableIntStateOf(0) }
    var delayRemainingMs by remember { mutableLongStateOf(0L) }
    var sessionStatus by remember { mutableStateOf("") }

    val scope = rememberCoroutineScope()

    Column(
        modifier = modifier
            .fillMaxSize()
            .padding(24.dp)
            .verticalScroll(rememberScrollState()),
        horizontalAlignment = Alignment.CenterHorizontally,
        verticalArrangement = Arrangement.Center,
    ) {
        when (step) {
            RecoveryStep.IDENTIFIER -> {
                // Icon + description
                Icon(
                    imageVector = Icons.Filled.Key,
                    contentDescription = null,
                    modifier = Modifier.size(64.dp),
                    tint = MaterialTheme.colorScheme.primary,
                )
                Spacer(Modifier.height(16.dp))
                Text(
                    text = stringResource(R.string.recovery_group_initiate_title),
                    style = MaterialTheme.typography.headlineSmall,
                    fontWeight = FontWeight.Bold,
                )
                Spacer(Modifier.height(8.dp))
                Text(
                    text = stringResource(R.string.recovery_group_initiate_description),
                    style = MaterialTheme.typography.bodyMedium,
                    color = MaterialTheme.colorScheme.onSurfaceVariant,
                    textAlign = TextAlign.Center,
                )
                Spacer(Modifier.height(24.dp))

                // Identifier input
                OutlinedTextField(
                    value = userIdentifier,
                    onValueChange = { userIdentifier = it },
                    label = { Text(stringResource(R.string.recovery_group_initiate_identifier)) },
                    singleLine = true,
                    keyboardOptions = KeyboardOptions(keyboardType = KeyboardType.Email),
                    modifier = Modifier
                        .fillMaxWidth()
                        .testTag("recovery-identifier-input"),
                )
                Spacer(Modifier.height(12.dp))

                // Hub input
                OutlinedTextField(
                    value = hubId,
                    onValueChange = { hubId = it },
                    label = { Text(stringResource(R.string.recovery_group_initiate_select_hub)) },
                    singleLine = true,
                    modifier = Modifier
                        .fillMaxWidth()
                        .testTag("recovery-hub-input"),
                )
                Spacer(Modifier.height(16.dp))

                // Error
                errorMessage?.let { error ->
                    Text(
                        text = error,
                        style = MaterialTheme.typography.bodySmall,
                        color = MaterialTheme.colorScheme.error,
                        modifier = Modifier.testTag("recovery-error"),
                    )
                    Spacer(Modifier.height(8.dp))
                }

                // Submit button
                Button(
                    onClick = {
                        scope.launch {
                            isSubmitting = true
                            errorMessage = null
                            try {
                                val deviceId = java.util.UUID.randomUUID().toString()
                                val encrypted = cryptoService.generateDeviceKeys(deviceId, "000000")
                                val pubkey = encrypted.state.encryptionPubkeyHex
                                val response = apiService.initiateRecovery(
                                    RecoveryInitiateRequest(
                                        hubId = hubId,
                                        userIdentifier = userIdentifier,
                                        newDevicePubkey = pubkey,
                                    ),
                                )
                                sessionId = response.sessionId
                                step = RecoveryStep.SIGNAL_VERIFICATION
                            } catch (e: Exception) {
                                errorMessage = e.message
                            }
                            isSubmitting = false
                        }
                    },
                    enabled = userIdentifier.isNotBlank() && hubId.isNotBlank() && !isSubmitting,
                    modifier = Modifier
                        .fillMaxWidth()
                        .testTag("start-recovery-button"),
                ) {
                    if (isSubmitting) {
                        CircularProgressIndicator(
                            modifier = Modifier.size(20.dp),
                            color = MaterialTheme.colorScheme.onPrimary,
                        )
                        Spacer(Modifier.size(8.dp))
                        Text(stringResource(R.string.recovery_group_initiate_submitting))
                    } else {
                        Text(stringResource(R.string.recovery_group_initiate_submit))
                    }
                }
            }

            RecoveryStep.SIGNAL_VERIFICATION -> {
                Icon(
                    imageVector = Icons.Filled.Shield,
                    contentDescription = null,
                    modifier = Modifier.size(64.dp),
                    tint = MaterialTheme.colorScheme.primary,
                )
                Spacer(Modifier.height(16.dp))
                Text(
                    text = stringResource(R.string.recovery_group_initiate_signal_verification),
                    style = MaterialTheme.typography.bodyMedium,
                    color = MaterialTheme.colorScheme.onSurfaceVariant,
                    textAlign = TextAlign.Center,
                )
                Spacer(Modifier.height(24.dp))

                OutlinedTextField(
                    value = verificationCode,
                    onValueChange = { verificationCode = it },
                    label = { Text(stringResource(R.string.recovery_group_initiate_verification_code)) },
                    singleLine = true,
                    keyboardOptions = KeyboardOptions(keyboardType = KeyboardType.Number),
                    modifier = Modifier
                        .fillMaxWidth()
                        .testTag("recovery-verification-code-input"),
                )
                Spacer(Modifier.height(16.dp))

                errorMessage?.let { error ->
                    Text(
                        text = error,
                        style = MaterialTheme.typography.bodySmall,
                        color = MaterialTheme.colorScheme.error,
                    )
                    Spacer(Modifier.height(8.dp))
                }

                Button(
                    onClick = {
                        scope.launch {
                            isSubmitting = true
                            errorMessage = null
                            try {
                                apiService.verifyRecoveryCode(
                                    RecoveryVerifyRequest(
                                        sessionId = sessionId,
                                        verificationCode = verificationCode,
                                    ),
                                )
                                step = RecoveryStep.WAITING
                            } catch (e: Exception) {
                                errorMessage = stringResource(R.string.recovery_group_error_signal_verification_failed)
                            }
                            isSubmitting = false
                        }
                    },
                    enabled = verificationCode.length >= 4 && !isSubmitting,
                    modifier = Modifier
                        .fillMaxWidth()
                        .testTag("verify-recovery-code-button"),
                ) {
                    if (isSubmitting) {
                        CircularProgressIndicator(
                            modifier = Modifier.size(20.dp),
                            color = MaterialTheme.colorScheme.onPrimary,
                        )
                    } else {
                        Text(stringResource(R.string.recovery_group_initiate_verify))
                    }
                }
            }

            RecoveryStep.WAITING -> {
                // Poll session status
                LaunchedEffect(sessionId) {
                    while (isActive) {
                        try {
                            val status = apiService.getRecoverySession(sessionId)
                            contributionCount = status.contributionCount
                            requiredApprovals = status.threshold
                            delayRemainingMs = status.delayRemainingMs ?: 0L
                            sessionStatus = status.status
                            if (status.status == "completed") {
                                // Decrypt contributions and restore PUK seed
                                step = RecoveryStep.COMPLETE
                                break
                            } else if (status.status in listOf("expired", "cancelled")) {
                                errorMessage = stringResource(R.string.recovery_group_error_session_expired)
                                step = RecoveryStep.IDENTIFIER
                                break
                            }
                        } catch (_: Exception) {
                            // Silently retry
                        }
                        delay(5_000L)
                    }
                }

                CircularProgressIndicator(
                    modifier = Modifier.size(48.dp),
                )
                Spacer(Modifier.height(24.dp))
                Text(
                    text = stringResource(R.string.recovery_group_initiate_waiting),
                    style = MaterialTheme.typography.titleMedium,
                    fontWeight = FontWeight.Bold,
                    textAlign = TextAlign.Center,
                )
                Spacer(Modifier.height(16.dp))

                // Approval progress
                Text(
                    text = "$contributionCount / $requiredApprovals",
                    style = MaterialTheme.typography.bodyMedium,
                    color = MaterialTheme.colorScheme.onSurfaceVariant,
                )
                Spacer(Modifier.height(8.dp))
                LinearProgressIndicator(
                    progress = {
                        contributionCount.toFloat() / maxOf(requiredApprovals, 1).toFloat()
                    },
                    modifier = Modifier
                        .fillMaxWidth()
                        .testTag("recovery-approval-progress"),
                )

                // Delay countdown
                if (delayRemainingMs > 0) {
                    Spacer(Modifier.height(16.dp))
                    val hours = delayRemainingMs / 3_600_000
                    val minutes = (delayRemainingMs % 3_600_000) / 60_000
                    val timeStr = if (hours > 0) "${hours}h ${minutes}m" else "${minutes}m"
                    Text(
                        text = stringResource(R.string.recovery_group_initiate_delay_countdown)
                            .replace("{{time}}", timeStr),
                        style = MaterialTheme.typography.bodyMedium,
                        color = MaterialTheme.colorScheme.tertiary,
                    )
                }
            }

            RecoveryStep.COMPLETE -> {
                Icon(
                    imageVector = Icons.Filled.CheckCircle,
                    contentDescription = null,
                    modifier = Modifier.size(64.dp),
                    tint = MaterialTheme.colorScheme.primary,
                )
                Spacer(Modifier.height(16.dp))
                Text(
                    text = stringResource(R.string.recovery_group_initiate_complete),
                    style = MaterialTheme.typography.headlineSmall,
                    fontWeight = FontWeight.Bold,
                )
                Spacer(Modifier.height(8.dp))
                Text(
                    text = stringResource(R.string.recovery_group_initiate_success),
                    style = MaterialTheme.typography.bodyMedium,
                    color = MaterialTheme.colorScheme.onSurfaceVariant,
                    textAlign = TextAlign.Center,
                )
                Spacer(Modifier.height(24.dp))
                Button(
                    onClick = { step = RecoveryStep.SET_PIN },
                    modifier = Modifier
                        .fillMaxWidth()
                        .testTag("recovery-set-pin-button"),
                ) {
                    Text(stringResource(R.string.recovery_group_initiate_set_pin))
                }
            }

            RecoveryStep.SET_PIN -> {
                Icon(
                    imageVector = Icons.Filled.Lock,
                    contentDescription = null,
                    modifier = Modifier.size(48.dp),
                    tint = MaterialTheme.colorScheme.primary,
                )
                Spacer(Modifier.height(16.dp))
                Text(
                    text = stringResource(R.string.recovery_group_initiate_set_pin),
                    style = MaterialTheme.typography.titleMedium,
                    fontWeight = FontWeight.Bold,
                    textAlign = TextAlign.Center,
                )
                Spacer(Modifier.height(24.dp))

                OutlinedTextField(
                    value = newPin,
                    onValueChange = { newPin = it },
                    label = { Text(stringResource(R.string.pin_enter_pin)) },
                    singleLine = true,
                    visualTransformation = PasswordVisualTransformation(),
                    keyboardOptions = KeyboardOptions(keyboardType = KeyboardType.NumberPassword),
                    modifier = Modifier
                        .fillMaxWidth()
                        .testTag("recovery-new-pin-input"),
                )
                Spacer(Modifier.height(12.dp))
                OutlinedTextField(
                    value = confirmPin,
                    onValueChange = { confirmPin = it },
                    label = { Text(stringResource(R.string.pin_confirm_pin)) },
                    singleLine = true,
                    visualTransformation = PasswordVisualTransformation(),
                    keyboardOptions = KeyboardOptions(keyboardType = KeyboardType.NumberPassword),
                    modifier = Modifier
                        .fillMaxWidth()
                        .testTag("recovery-confirm-pin-input"),
                )
                Spacer(Modifier.height(16.dp))

                errorMessage?.let { error ->
                    Text(
                        text = error,
                        style = MaterialTheme.typography.bodySmall,
                        color = MaterialTheme.colorScheme.error,
                    )
                    Spacer(Modifier.height(8.dp))
                }

                Button(
                    onClick = {
                        if (newPin == confirmPin && newPin.length >= 6) {
                            onRecoveryComplete()
                        } else {
                            errorMessage = "PINs do not match or too short"
                        }
                    },
                    enabled = newPin.length >= 6 && newPin == confirmPin,
                    modifier = Modifier
                        .fillMaxWidth()
                        .testTag("recovery-save-pin-button"),
                ) {
                    Text(stringResource(R.string.action_save))
                }
            }
        }
    }
}

private enum class RecoveryStep {
    IDENTIFIER,
    SIGNAL_VERIFICATION,
    WAITING,
    COMPLETE,
    SET_PIN,
}
```

**Note:** String resources referenced as `R.string.recovery_group_*` are generated by `bun run i18n:codegen` from the `recoveryGroup.*` keys in en.json. The codegen flattens nested keys with underscores: `recoveryGroup.initiate.title` becomes `recovery_group_initiate_title` in `strings.xml`. Also references `R.string.pin_enter_pin`, `R.string.pin_confirm_pin`, `R.string.action_save` which already exist in the codebase.
